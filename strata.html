Smart Contract & Architecture

Inspect the underlying Solidity smart contract code, gas metrics, and consensus architecture.

CollegeFeePayment.sol Solidity ^0.8.20 
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

/**
 * @title CollegeFeePayment
 * @dev Academic fee payment contract utilizing Checks-Effects-Interactions (CEI) pattern.
 */
contract CollegeFeePayment {
    address public owner;
    address payable public collegeAccount;
    uint256 public feeAmount = 0.05 ether;

    struct PaymentRecord {
        bool hasPaid;
        uint256 amountPaid;
        uint256 timestamp;
        string studentId;
        string studentName;
        string department;
        string semester;
    }

    mapping(address => PaymentRecord) public records;
    address[] public studentList;

    event FeePaid(
        address indexed studentAddress,
        string studentId,
        string studentName,
        uint256 amount,
        uint256 timestamp
    );

    constructor(address payable _collegeAccount) {
        owner = msg.sender;
        collegeAccount = _collegeAccount;
    }

    function payFee(
        string memory _id,
        string memory _name,
        string memory _dept,
        string memory _sem
    ) public payable {
        // 1. CHECKS
        require(msg.value == feeAmount, "Incorrect fee amount");
        require(!records[msg.sender].hasPaid, "Fee already paid for this wallet");

        // 2. EFFECTS
        records[msg.sender] = PaymentRecord(true, msg.value, block.timestamp, _id, _name, _dept, _sem);
        studentList.push(msg.sender);

        // 3. INTERACTIONS (Low-level call prevents 2300 gas stipend crash)
        (bool success, ) = collegeAccount.call{value: msg.value}("");
        require(success, "Transfer to college failed");

        emit FeePaid(msg.sender, _id, _name, msg.value, block.timestamp);
    }
}

Examiner Viva Defense Guide

Q1: Why is the Checks-Effects-Interactions (CEI) pattern enforced?
The CEI pattern checks preconditions, modifies the internal state storage (records[msg.sender].hasPaid = true), and only then sends Ether. This eliminates Reentrancy attacks where a malicious contract could recursively drain funds.
Q2: Why use .call over .transfer()?
The .transfer() method forwards a hardcoded 2300 gas stipend, which fails if the college account is a multi-signature wallet or smart contract. Modern Solidity best practice mandates using .call with a boolean success check.

Frequently Asked Questions

Common questions regarding blockchain tuition payments, wallet setups, and clearance proofs.

How do I pay fees using the Sepolia Testnet? ▼
Install the MetaMask extension in your browser, switch your network to "Sepolia", get free testnet ETH from a Sepolia faucet (e.g. Google Cloud Web3 Faucet), and click "Pay with MetaMask" on our portal!
What if I don't have MetaMask installed? ▼
Our dApp includes an interactive Demo Sandbox Mode. You can test and demonstrate the complete fee payment, mining, and certificate generation flow seamlessly during your presentation without needing an active wallet extension.
Can a student pay twice from the same wallet? ▼
No. The smart contract validates require(!records[msg.sender].hasPaid, "Fee already paid"), preventing duplicate charges and double payments.
How is my official fee receipt verified? ▼
Every payment generates an Ethereum transaction hash stored permanently in the blockchain's receipt trie. Anyone can verify clearance instantly on our "Verify Clearance" page using just your Roll ID.
